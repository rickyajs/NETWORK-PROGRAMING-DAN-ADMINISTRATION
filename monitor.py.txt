import subprocess
import time
import requests
from datetime import datetime

# ==================== KONFIGURASI ====================
# Target host yang sedang digunakan (Router / Google DNS)
TARGET_HOST = "192.168.1.1"  

# Token API Fonnte Anda
FONNTE_TOKEN = "w36jRS5cDj1ixS7VNGvd"

# Nomor Penerima WA
TARGET_WA = "081265697722"

CHECK_INTERVAL = 5
MAX_FAIL_COUNT = 3  
# ====================================================

def send_whatsapp_notif(status, host):
    url = "https://api.fonnte.com/send"
    headers = {"Authorization": FONNTE_TOKEN}
    now = datetime.now().strftime("%d-%m-%Y %H:%M:%S")

    if status == "DOWN":
        message = (
            f"🚨 *PERHATIAN: JARINGAN INTERNET ANDA SEDANG DOWN!* 🚨\n\n"
            f"Target IP: *{host}*\n"
            f"Status: OFFLINE / UNREACHABLE\n"
            f"Waktu: {now}\n\n"
            f"⚠️ KONEKSI JARINGAN ANDA SEDANG TERPUTUS! MOHON SEGERA CEK JARINGAN ANDA."
        )
    else:
        message = (
            f"✅ *PERHATIAN: JARINGAN ANDA SUDAH KEMBALI NORMAL* ✅\n\n"
            f"Target IP: *{host}*\n"
            f"Status: ONLINE / NORMAL\n"
            f"Waktu: {now}\n\n"
            f"KONEKSI JARINGAN ANDA SUDAH KEMBALI TERHUBUNG."
        )

    payload = {
        "target": TARGET_WA,
        "message": message,
        "countryCode": "62"
    }

    # Mekanisme Retry jika gagal kirim WA
    for attempt in range(1, 4):
        try:
            print(f"[{now}] Mengirim WA ({status}) - Percobaan ke-{attempt}...")
            response = requests.post(url, headers=headers, data=payload, timeout=15)
            res_data = response.json()
            
            print(f"[{now}] Respon Fonnte: {res_data}")
            
            # Cek status sukses dari Fonnte
            if res_data.get("status") == True:
                print(f"[{now}] ✅ Pesan WhatsApp BERHASIL terkirim!")
                break
            else:
                print(f"[{now}] ❌ Fonnte menolak mengirim pesan. Detail: {res_data.get('reason')}")
        except Exception as e:
            print(f"[{now}] ❌ Error koneksi ke Fonnte (Attempt {attempt}): {e}")
        
        time.sleep(2) # Jeda sebelum coba lagi

def ping_windows(host):
    cmd = ["ping", "-n", "1", "-w", "1000", host]
    return subprocess.call(cmd, stdout=subprocess.DEVNULL, stderr=subprocess.DEVNULL) == 0

def start_monitoring():
    print("==================================================")
    print("                MONITORING JARINGAN               ")
    print("==================================================")
    print(f"Target Host : {TARGET_HOST}")
    print(f"Penerima WA : {TARGET_WA}\n")

    is_down = False
    fail_counter = 0

    while True:
        timestamp = datetime.now().strftime("%H:%M:%S")
        is_reachable = ping_windows(TARGET_HOST)

        if is_reachable:
            if is_down:
                print(f"[{timestamp}] ✅ Connection Restored to {TARGET_HOST}")
                send_whatsapp_notif("UP", TARGET_HOST)
                is_down = False
            else:
                print(f"[{timestamp}] 🟢 Koneksi OK")
            
            fail_counter = 0
        else:
            fail_counter += 1
            print(f"[{timestamp}] ⚠️ Ping Gagal ({fail_counter}/{MAX_FAIL_COUNT})")

            if fail_counter >= MAX_FAIL_COUNT and not is_down:
                print(f"[{timestamp}] 🚨 Jaringan terdeteksi DOWN! Memproses Pengriman Pesan Ke WhatsApp...")
                send_whatsapp_notif("DOWN", TARGET_HOST)
                is_down = True

        time.sleep(CHECK_INTERVAL)

if __name__ == "__main__":
    start_monitoring()
